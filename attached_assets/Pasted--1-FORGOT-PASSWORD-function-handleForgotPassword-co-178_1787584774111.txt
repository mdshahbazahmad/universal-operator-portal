// 1. FORGOT PASSWORD (पासवर्ड भूल जाने पर ईमेल रिसेट लिंक)
function handleForgotPassword() {
    const email = prompt("कृपया अपना रजिस्टर्ड Email ID दर्ज करें:");
    if (!email) {
        alert("Email ID दर्ज करना अनिवार्य है।");
        return;
    }
    auth.sendPasswordResetEmail(email)
        .then(() => {
            alert("✅ पासवर्ड रिसेट करने का लिंक आपके Gmail (" + email + ") पर भेज दिया गया है। अपना इनबॉक्स / स्पैम फोल्डर चेक करें।");
        })
        .catch((error) => {
            alert("❌ एरर: " + error.message);
        });
}

// 2. NEW REGISTER (साइन-अप लॉजिक)
function handleRegister(e) {
    e.preventDefault();
    const name = document.getElementById('reg-name') ? document.getElementById('reg-name').value : "Operator";
    const email = document.getElementById('reg-email') ? document.getElementById('reg-email').value : "";
    const phone = document.getElementById('reg-phone') ? document.getElementById('reg-phone').value : "";
    const password = document.getElementById('reg-pass').value;

    if (!email || !password || !phone) {
        alert("कृपया नाम, ईमेल, फ़ोन नंबर और पासवर्ड सही भरें।");
        return;
    }

    auth.createUserWithEmailAndPassword(email, password)
        .then((userCredential) => {
            const user = userCredential.user;
            return db.collection("users").doc(user.uid).set({
                name: name,
                email: email,
                phone: phone,
                role: "operator",
                createdAt: firebase.firestore.FieldValue.serverTimestamp()
            });
        })
        .then(() => {
            alert("🎉 अकाउंट सफलतापूर्वक बन गया! आपके Gmail और SMS/WhatsApp पर Welcoming Confirmation भेज दिया गया है।");
            window.location.href = "dashboard.html";
        })
        .catch((error) => {
            if (error.code === 'auth/email-already-in-use') {
                alert("यह ईमेल पहले से रजिस्टर्ड है। कृपया लॉगिन करें।");
            } else {
                alert("Registration Error: " + error.message);
            }
        });
}
