<script>
  function toggleAiChat() {
    const box = document.getElementById('ai-chat-box');
    box.style.display = (box.style.display === 'flex') ? 'none' : 'flex';
  }

  function handleKeyPress(e) {
    if (e.key === 'Enter') processUserChat();
  }

  function sendQuickMessage(text) {
    document.getElementById('user-chat-input').value = text;
    processUserChat();
  }

  function processUserChat() {
    const input = document.getElementById('user-chat-input');
    const query = input.value.trim();
    if (!query) return;

    appendMessage(query, 'user');
    input.value = '';

    // Show Typing Indicator
    const typingDiv = document.createElement('div');
    typingDiv.className = 'typing-indicator';
    typingDiv.id = 'typing-indicator';
    typingDiv.innerText = 'AI सोच रहा है...';
    document.getElementById('chat-messages').appendChild(typingDiv);
    scrollChatToBottom();

    // Generate AI Response after 600ms delay
    setTimeout(() => {
      document.getElementById('typing-indicator')?.remove();
      const botResponse = generateAiResponse(query);
      appendMessage(botResponse, 'bot');
    }, 600);
  }

  function appendMessage(text, sender) {
    const chatBody = document.getElementById('chat-messages');
    const msgDiv = document.createElement('div');
    msgDiv.className = `msg-bubble msg-${sender}`;
    msgDiv.innerHTML = text;
    chatBody.appendChild(msgDiv);
    scrollChatToBottom();
  }

  function scrollChatToBottom() {
    const chatBody = document.getElementById('chat-messages');
    chatBody.scrollTop = chatBody.scrollHeight;
  }

  // Comprehensive AI Response Logic
  function generateAiResponse(q) {
    const input = q.toLowerCase();

    if (input.includes('पैसे') || input.includes('कटेंगे') || input.includes('रिचार्ज') || input.includes('payment') || input.includes('recharge') || input.includes('मुनाफा')) {
      return `💵 <b>रिचार्ज एवं पेमेंट की स्पष्ट प्रक्रिया:</b><br><br>
      <b>1. ग्राहक भुगतान:</b> कस्टमर ऐप से जो भी ₹500/₹600 का रिचार्ज करेगा, 100% पैसा <b>सीधे आपके बैंक खाते/पेमेंट गेटवे</b> में आएगा।<br>
      <b>2. सर्वर कटौती:</b> हमारा यूनिवर्सल पोर्टल आपके मुख्य सर्वर को API सिग्नल देगा। मुख्य सर्वर आपके वॉलेट से केवल अपनी लागत (जैसे ₹150) काटेगा।<br>
      <b>3. आपका मुनाफा:</b> ₹500 - ₹150 = <b>₹350 का शुद्ध मुनाफा</b> सीधे आपके पास रहेगा!`;
    }

    if (input.includes('दो पोर्टल') || input.includes('मुख्य') || input.includes('server') || input.includes('mikrotik') || input.includes('radius') || input.includes('अलग')) {
      return `🖥️ <b>दो अलग पोर्टल कैसे काम करते हैं?</b><br><br>
      • आपका मुख्य पोर्टल (Mikrotik / Radius / OLT) बैकएंड सर्वर की तरह इंटरनेट कंट्रोल करता है।<br>
      • हमारा <b>यूनिवर्सल पोर्टल</b> फ्रंटएंड ऐप/वेबसाइट है जो ग्राहकों को दिखता है।<br>
      • दोनों पोर्टल आपस में <b>API ब्रिज</b> से जुड़े रहते हैं। ग्राहक यूनिवर्सल पोर्टल पर रिचार्ज करेगा और इंटरनेट आपके मुख्य सर्वर पर चालू हो जाएगा!`;
    }

    if (input.includes('डेटा') || input.includes('सुरक्षा') || input.includes('security') || input.includes('data') || input.includes('गोपनीय')) {
      return `🔒 <b>डेटा एवं सुरक्षा गारंटी:</b><br><br>
      • पूरा नेटवर्क <b>256-Bit SSL/TLS एन्क्रिप्शन</b> से सुरक्षित है।<br>
      • आपके ग्राहकों और वित्तीय लेन-देन का डेटा 100% गोपनीय है और केवल आपकी संपत्ति है।<br>
      • किसी अन्य ऑपरेटर के पास आपका डेटा नहीं जा सकता।`;
    }

    if (input.includes('ऐप') || input.includes('app') || input.includes('डाउनलोड') || input.includes('कस्टमर')) {
      return `📱 <b>कस्टमर ऐप की सुविधा:</b><br><br>
      • ग्राहक आपकी ब्रांडेड ऐप से 24x7 रिचार्ज कर सकते हैं, सिग्नल फॉल्ट की शिकायत दर्ज कर सकते हैं और बिल डाउनलोड कर सकते हैं।<br>
      • ऐप में केवल आपकी कंपनी का नाम, लोगो और बैनर दिखेगा।`;
    }

    return `🤖 <b>Eagle Network Support:</b><br><br>
    आपकी सेवा के लिए हमारा सिस्टम 24x7 एक्टिव है।<br><br>
    क्या आप इनमें से कुछ पूछना चाहते हैं?<br>
    • <b>'पैसे कहाँ जाएँगे'</b><br>
    • <b>'सर्वर सिंक कैसे होगा'</b><br>
    • <b>'कस्टमर ऐप कैसे काम करेगी'</b><br><br>
    आप चाहें तो हमारी सपोर्ट टीम से सीधे WhatsApp पर भी बात कर सकते हैं।`;
  }
</script>
